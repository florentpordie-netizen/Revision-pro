    function loadCurrentSession() {
        try {
            const session = localStorage.getItem('current-session');
            if (session) {
                const email = JSON.parse(session);
                if (users[email]) {
                    state.currentUser = email;
                    state.themes = users[email].themes || [];
                    state.appName = users[email].appName || 'Révision Pro';
                    state.revisionPeriod = users[email].revisionPeriod || 'custom';
                    state.darkMode = users[email].darkMode || false;

                }

        } catch (error) {}

    function saveCurrentSession() {
        if (state.currentUser) {
            users[state.currentUser] = {
                email: state.currentUser,
                password: users[state.currentUser].password,
                themes: state.themes,
                appName: state.appName,
                revisionPeriod: state.revisionPeriod,
                darkMode: state.darkMode,
                statistics: state.statistics
            };
            saveUsers();
            localStorage.setItem('current-session', JSON.stringify(state.currentUser));
        }
    }

    function updateStatistics(action) {
        if (!state.statistics.totalReviews) state.statistics.totalReviews = 0;
        if (!state.statistics.correctAnswers) state.statistics.correctAnswers = 0;
        if (!state.statistics.streak) state.statistics.streak = 0;
        if (!state.statistics.lastReviewDate) state.statistics.lastReviewDate = null;

        if (action === 'correct') {
            state.statistics.totalReviews++;
            state.statistics.correctAnswers++;
            
            const today = new Date().toDateString();
            if (state.statistics.lastReviewDate !== today) {
                state.statistics.streak++;
            }
            state.statistics.lastReviewDate = today;
        } else if (action === 'incorrect') {
            state.statistics.totalReviews++;
        }

        saveCurrentSession();
    }

    function login(email, password) {
        if (!email || !password) return { success: false, message: 'Veuillez remplir tous les champs' };
        if (!email.toLowerCase().endsWith('@gmail.com')) return { success: false, message: 'Veuillez utiliser Gmail' };

        if (users[email]) {
            if (users[email].password === password) {
                state.currentUser = email;
                state.themes = users[email].themes || [];
                state.appName = users[email].appName || 'Révision Pro';
                state.revisionPeriod = users[email].revisionPeriod || 'custom';
                state.darkMode = users[email].darkMode || false;
                state.statistics = users[email].statistics || {};
                if (state.darkMode) document.body.classList.add('dark-mode');
                localStorage.setItem('current-session', JSON.stringify(email));
                requestNotificationPermission();
                return { success: true };
            }
            return { success: false, message: 'Mot de passe incorrect' };
        }
        return { success: false, message: 'Compte introuvable' };
    }

    function register(email, password, confirmPassword) {
        if (!email || !password || !confirmPassword) return { success: false, message: 'Remplir tous les champs' };
        if (!email.toLowerCase().endsWith('@gmail.com')) return { success: false, message: 'Utiliser Gmail' };
        if (password.length < 6) return { success: false, message: 'Min. 6 caractères' };
        if (password !== confirmPassword) return { success: false, message: 'Mots de passe différents' };
        if (users[email]) return { success: false, message: 'Compte existant' };

        users[email] = {

            themes: initialThemes,

    function getIntervals() {
        const period = revisionPeriods.find(p => p.id === state.revisionPeriod);
        return period ? period.intervals.map(d => d * 24 * 60 * 60 * 1000) : [86400000, 259200000, 604800000, 1209600000, 2592000000];
    }

    function getNextInterval(box) {
        const intervals = getIntervals();
        return intervals[box - 1] || intervals[0];
    }

    function getCardsToReview(themeId) {
        const theme = state.themes.find(t => t.id === themeId);
        if (!theme) return [];
        return theme.cards.filter(card => card.nextReview <= Date.now()).sort((a, b) => a.nextReview - b.nextReview);
    }

    function reviewCard(known) {
        state.themes = state.themes.map(theme => {
            if (theme.id === state.selectedTheme.id) {
                return {
                    ...theme,
                    cards: theme.cards.map(card => {
                        if (card.id === state.currentCard.id) {
                            const newBox = known ? Math.min(card.box + 1, 5) : 1;
                            return {
                                ...card,
                                box: newBox,
                                lastReview: Date.now(),
                                nextReview: Date.now() + getNextInterval(newBox),
                                reviewCount: (card.reviewCount || 0) + 1
                            };
                        }
                        return card;
                    })
                };
            }
            return theme;
        });

        updateStatistics(known ? 'correct' : 'incorrect');
        saveCurrentSession();
        state.showAnswer = false;
        
        const nextCards = getCardsToReview(state.selectedTheme.id);
        if (nextCards.length > 0) {
            state.currentCard = nextCards[0];
        } else {
            state.currentCard = null;
            state.currentView = 'theme';
            showNotification('🎉 Terminé !');
        }
        
        render();
    }

    function addCard() {

        const answer = document.getElementById('cardAnswer').value.trim();
        if (!question || !answer) return;

        state.themes = state.themes.map(theme => {

        state.selectedTheme = state.themes.find(t => t.id === state.selectedTheme.id);
        document.getElementById('cardQuestion').value = '';
        document.getElementById('cardAnswer').value = '';
        showNotification('Carte ajoutée !');
        render();
    }

    function addBulkCards() {
        const bulkText = document.getElementById('bulkCards').value.trim();
        if (!bulkText) return;

        const newCards = [];
        bulkText.split('\n').forEach((line, i) => {
            line = line.trim();
            if (!line) return;
            const parts = line.split('|');
            if (parts.length === 2) {
                newCards.push({
                    id: Date.now() + i,
                    question: parts[0].trim(),
                    answer: parts[1].trim(),
                    box: 1,
                    lastReview: null,
                    nextReview: Date.now(),
                    reviewCount: 0
                });
            }
        });

        if (newCards.length === 0) {
            showNotification('Format: Question | Réponse', 'error');
            return;
        }

        state.themes = state.themes.map(theme => {
            if (theme.id === state.selectedTheme.id) {
                return { ...theme, cards: [...theme.cards, ...newCards] };
            }
            return theme;
        });

        saveCurrentSession();
        state.selectedTheme = state.themes.find(t => t.id === state.selectedTheme.id);
        document.getElementById('bulkCards').value = '';
        showNotification(`${newCards.length} carte(s) ajoutée(s) !`);
        render();
    }

    function deleteCard(cardId) {
        if (!confirm('Supprimer ?')) return;
        state.themes = state.themes.map(theme => {
            if (theme.id === state.selectedTheme.id) {
                return { ...theme, cards: theme.cards.filter(c => c.id !== cardId) };
            }
            return theme;
        });
        saveCurrentSession();
        state.selectedTheme = state.themes.find(t => t.id === state.selectedTheme.id);
        showNotification('Supprimé');
        render();
    }

    function deleteTheme(themeId) {

        saveCurrentSession();
        showNotification('Thématique supprimée');
        render();
    }

    function importData() {
        const file = document.getElementById('importFile').files[0];
        if (!file) {
            showNotification('Sélectionner un fichier', 'error');
            return;
        }

        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const data = JSON.parse(e.target.result);
                if (confirm('Remplacer toutes les données ?')) {
                    state.themes = data.themes || [];
                    if (data.appName) state.appName = data.appName;
                    saveCurrentSession();
                    showNotification('Données importées !');
                    closeModal('importModal');
                    render();
                }
            } catch (error) {
                showNotification('Fichier invalide', 'error');
            }
        };
        reader.readAsText(file);
    }

    function generateShareCode(themeId) {
        const theme = state.themes.find(t => t.id === themeId);
        if (!theme) return;

        const shareData = {
            name: theme.name,
            color: theme.color,
            cards: theme.cards.map(c => ({ question: c.question, answer: c.answer }))
        };

        const code = btoa(JSON.stringify(shareData)).substring(0, 8).toUpperCase();
        const shares = JSON.parse(localStorage.getItem('shared-themes') || '{}');
        shares[code] = shareData;
        localStorage.setItem('shared-themes', JSON.stringify(shares));

        openModal('shareModal');
        document.getElementById('shareCode').textContent = code;
    }

    function importSharedTheme() {
        const code = document.getElementById('shareCodeInput').value.trim().toUpperCase();
        if (!code) {
            showNotification('Entrer un code', 'error');
            return;
        }

        const shares = JSON.parse(localStorage.getItem('shared-themes') || '{}');
        const sharedTheme = shares[code];

        if (!sharedTheme) {
            showNotification('Code invalide', 'error');
            return;
        }

        const newTheme = {
            id: Date.now(),
            name: sharedTheme.name,
            color: sharedTheme.color,
            cards: sharedTheme.cards.map((c, i) => ({
                id: Date.now() + i,
                question: c.question,
                answer: c.answer,
                box: 1,
                lastReview: null,
                nextReview: Date.now(),
                reviewCount: 0
            }))
        };

        state.themes.push(newTheme);
        saveCurrentSession();
        showNotification(`"${newTheme.name}" importée !`);
        closeModal('shareModal');
        render();
    }
